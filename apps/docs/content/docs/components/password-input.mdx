Masked input with accessible show/hide controls, password-manager hints and validation in both input styles.

## Complete example

The demo and this example use the same implementation. Reuse the existing provider when integrating this into an app.

```tsx title="PasswordinputsExample.tsx"
import * as React from 'react';
import { View, ScrollView } from 'react-native';
import { SafeAreaProvider } from 'react-native-safe-area-context';
import { GestureHandlerRootView } from 'react-native-gesture-handler';
import * as UI from '@isoline-ui/react-native';

type StoryProps = { de?: boolean };

function PasswordExample({ de = false }: StoryProps) {
  const [password, setPassword] = React.useState('');
  const [confirmation, setConfirmation] = React.useState('');
  const theme = UI.useIsolineTheme();
  const t = (en: string, german: string) => (de ? german : en);
  return (
    <UI.IsolineProvider
      theme={theme}
      style={{ flex: 0 }}
      labels={{
        showPassword: t('Show password', 'Passwort anzeigen'),
        hidePassword: t('Hide password', 'Passwort verbergen'),
      }}
    >
      <View style={{ gap: 20 }}>
        <View testID="password-text-field">
          <UI.TextField
            label={t('New password', 'Neues Passwort')}
            value={password}
            onChangeText={setPassword}
            secureTextEntry
            notchClassName="bg-background"
            autoComplete="new-password"
            textContentType="newPassword"
            error={
              password.length > 0 && password.length < 8
                ? t(
                    'Example: use at least 8 characters.',
                    'Beispiel: mindestens 8 Zeichen verwenden.',
                  )
                : undefined
            }
          />
        </View>
        <View testID="password-basic-input">
          <UI.Input
            label={t('Account password', 'Kontopasswort')}
            value={confirmation}
            onChangeText={setConfirmation}
            secureTextEntry
            autoComplete="current-password"
            textContentType="password"
            hint={t(
              'Use the eye button to show or hide the text.',
              'Mit dem Auge den Text anzeigen oder verbergen.',
            )}
          />
        </View>
        <UI.Text>
          {t(
            'Both input styles keep their value when visibility changes. Password fields remain solid form content.',
            'Beide Eingabestile behalten ihren Wert beim Umschalten. Passwortfelder bleiben solide Formularinhalte.',
          )}
        </UI.Text>
      </View>
    </UI.IsolineProvider>
  );
}

export function Example() {
  return (
    <GestureHandlerRootView style={{ flex: 1 }}>
      <SafeAreaProvider>
        <UI.IsolineProvider theme={UI.themes.violet.light}>
          <PasswordExample de={false} />
        </UI.IsolineProvider>
      </SafeAreaProvider>
    </GestureHandlerRootView>
  );
}
```

## API

Public components: `Input`, `TextField`.

Key props: `secureTextEntry`, `autoComplete`, `textContentType`, `value`, `onChangeText`, `editable`, `error`. Exported props types define the full contract. Labels and actions are supplied by the app.

Check large text, disabled states and VoiceOver/TalkBack in the native screen. Browser demos do not replace device tests.
