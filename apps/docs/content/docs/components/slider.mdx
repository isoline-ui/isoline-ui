An adjustable numeric value with pointer, keyboard and accessibility actions.

## Usage and states

The default range is 0–1; `step=0` is continuous. Values are clamped and discrete steps quantized. `onValueChange` updates during dragging and accessibility/keyboard adjustments; `onSlidingComplete` is for committing a gesture. Supply a human-readable `valueLabel` for units and an accessible control label. Disabled sliders cannot change. Colors can be overridden individually; `style` targets the outer wrapper.

## Complete example

The demo and this example use the same implementation. Reuse the existing provider when integrating this into an app.

```tsx title="SliderExample.tsx"
import * as React from 'react';
import { View, ScrollView } from 'react-native';
import { SafeAreaProvider } from 'react-native-safe-area-context';
import { GestureHandlerRootView } from 'react-native-gesture-handler';
import * as UI from '@isoline-ui/react-native';

type StoryProps = { de?: boolean };

function SliderExample({ de = false }: StoryProps) {
  const [value, setValue] = React.useState(40);
  const t = (en: string, german: string) => (de ? german : en);
  return (
    <View style={{ gap: 20 }}>
      <UI.Slider
        value={value}
        onValueChange={setValue}
        minimumValue={0}
        maximumValue={100}
        step={5}
        accessibilityLabel={t('Intensity', 'Intensität')}
        valueLabel={`${value}%`}
        leadingLabel="0%"
        trailingLabel="100%"
      />
      <UI.Text>{value}%</UI.Text>
      <UI.Slider
        value={25}
        onValueChange={() => {}}
        disabled
        accessibilityLabel={t('Unavailable slider', 'Nicht verfügbarer Regler')}
        minimumValue={0}
        maximumValue={100}
      />
    </View>
  );
}

export function Example() {
  return (
    <GestureHandlerRootView style={{ flex: 1 }}>
      <SafeAreaProvider>
        <UI.IsolineProvider theme={UI.themes.violet.light}>
          <SliderExample de={false} />
        </UI.IsolineProvider>
      </SafeAreaProvider>
    </GestureHandlerRootView>
  );
}
```
