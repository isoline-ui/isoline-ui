Choose one visible option with keyboard and screen-reader support.

## Usage and states

Use this for one visible choice. Put uniquely valued `RadioItem` children directly inside `RadioGroup`; fragments and nested groups are not registered. The group is controlled and accepts `null` for no selection. Disabled items are skipped by arrow keys; Home/End select the first/last available item. Supply a group label. Use Checkbox for independent multiple choices.

## Complete example

The demo and this example use the same implementation. Reuse the existing provider when integrating this into an app.

```tsx title="RadiogroupExample.tsx"
import * as React from 'react';
import { View, ScrollView } from 'react-native';
import { SafeAreaProvider } from 'react-native-safe-area-context';
import { GestureHandlerRootView } from 'react-native-gesture-handler';
import * as UI from '@isoline-ui/react-native';

type StoryProps = { de?: boolean };

function RadioExample({ de = false }: StoryProps) {
  const [value, setValue] = React.useState<string | null>('standard');
  return (
    <UI.RadioGroup
      value={value}
      onValueChange={setValue}
      accessibilityLabel={de ? 'Versandart' : 'Delivery'}
    >
      <UI.RadioItem
        value="standard"
        label={de ? 'Standardversand' : 'Standard delivery'}
        description={de ? 'Drei bis fünf Tage' : 'Three to five days'}
      />
      <UI.RadioItem value="express" label="Express" />
      <UI.RadioItem
        value="unavailable"
        label={de ? 'Nicht verfügbar' : 'Unavailable'}
        disabled
      />
    </UI.RadioGroup>
  );
}

export function Example() {
  return (
    <GestureHandlerRootView style={{ flex: 1 }}>
      <SafeAreaProvider>
        <UI.IsolineProvider theme={UI.themes.violet.light}>
          <RadioExample de={false} />
        </UI.IsolineProvider>
      </SafeAreaProvider>
    </GestureHandlerRootView>
  );
}
```
