Reveal controlled content without a navigation change.

## Usage and states

Keep `expanded` in app state. Each instance is an independent disclosure; enforce single-open behavior in your parent if needed. Hidden content unmounts by default. `keepMounted` retains local state while removing collapsed content from layout and focus order. The trigger announces expanded/disabled states. `className` and `style` apply to the outer group.

## Complete example

The demo and this example use the same implementation. Reuse the existing provider when integrating this into an app.

```tsx title="AccordionExample.tsx"
import * as React from 'react';
import { View, ScrollView } from 'react-native';
import { SafeAreaProvider } from 'react-native-safe-area-context';
import { GestureHandlerRootView } from 'react-native-gesture-handler';
import * as UI from '@isoline-ui/react-native';

type StoryProps = { de?: boolean };

function AccordionExample({ de = false }: StoryProps) {
  const [expanded, setExpanded] = React.useState(false);
  return (
    <UI.Accordion
      title={de ? 'Weitere Informationen' : 'More information'}
      expanded={expanded}
      onExpandedChange={setExpanded}
    >
      <UI.Text>
        {de ? 'Dieser Inhalt wird auf Wunsch sichtbar.' : 'This content appears on request.'}
      </UI.Text>
      <UI.Button label={de ? 'Schließen' : 'Collapse'} onPress={() => setExpanded(false)} />
    </UI.Accordion>
  );
}

export function Example() {
  return (
    <GestureHandlerRootView style={{ flex: 1 }}>
      <SafeAreaProvider>
        <UI.IsolineProvider theme={UI.themes.violet.light}>
          <AccordionExample de={false} />
        </UI.IsolineProvider>
      </SafeAreaProvider>
    </GestureHandlerRootView>
  );
}
```
