Search a local or asynchronously supplied single-selection list.

## Usage and states

Use for longer single-selection lists. `null` displays the placeholder. Local mode filters option labels without case sensitivity; disabled results cannot be selected. The sheet contains a bounded virtualized list. For remote search, supply both `query` and `onQueryChange`; `options` must then already contain the results. The app must debounce requests, cancel or ignore stale responses, and retain the selected option when changing results. `loading` replaces results; use localized `loadingLabel`, `searchLabel` and `emptyLabel`. `error` is a field error, not an automatic retry mechanism.

## Complete example

The demo and this example use the same implementation. Reuse the existing provider when integrating this into an app.

```tsx title="ComboboxExample.tsx"
import * as React from 'react';
import { View, ScrollView } from 'react-native';
import { SafeAreaProvider } from 'react-native-safe-area-context';
import { GestureHandlerRootView } from 'react-native-gesture-handler';
import * as UI from '@isoline-ui/react-native';

type StoryProps = { de?: boolean };

function ComboboxExample({ de = false }: StoryProps) {
  const [value, setValue] = React.useState<string | null>(null);
  return (
    <UI.Combobox
      value={value}
      onChange={setValue}
      label={de ? 'Land' : 'Country'}
      accessibilityLabel={de ? 'Land auswählen' : 'Choose country'}
      placeholder={de ? 'Bitte auswählen' : 'Select a country'}
      searchLabel={de ? 'Länder suchen' : 'Search countries'}
      emptyLabel={de ? 'Keine Treffer' : 'No results'}
      options={[
        { id: 'de', label: de ? 'Deutschland' : 'Germany' },
        { id: 'fr', label: de ? 'Frankreich' : 'France' },
        { id: 'it', label: de ? 'Italien' : 'Italy' },
        { id: 'unavailable', label: de ? 'Nicht verfügbar' : 'Unavailable', disabled: true },
      ]}
    />
  );
}

export function Example() {
  return (
    <GestureHandlerRootView style={{ flex: 1 }}>
      <SafeAreaProvider>
        <UI.IsolineProvider theme={UI.themes.violet.light}>
          <ComboboxExample de={false} />
        </UI.IsolineProvider>
      </SafeAreaProvider>
    </GestureHandlerRootView>
  );
}
```
