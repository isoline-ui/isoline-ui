Keep localized editing text separate from parsed numeric units. Currency, validation, rounding and persistence remain explicit caller decisions.

## Usage and states

Keep `value` as the editing string so incomplete input is not lost. `onAmountChange` reports a number in major currency units or null for incomplete/invalid text, not integer cents. `locale` controls decimal/group separators and `currency` supplies a symbol; neither performs conversion. Check finite values and product limits before saving. In financial calculations use an appropriate decimal/minor-unit strategy in your app.

## Complete example

The demo and this example use the same implementation. Reuse the existing provider when integrating this into an app.

```tsx title="AmountfieldExample.tsx"
import * as React from 'react';
import { View, ScrollView } from 'react-native';
import { SafeAreaProvider } from 'react-native-safe-area-context';
import { GestureHandlerRootView } from 'react-native-gesture-handler';
import * as UI from '@isoline-ui/react-native';

type StoryProps = { de?: boolean };

function AmountFieldExample({ de = false }: StoryProps) {
  const [value, setValue] = React.useState(de ? '12,90' : '12.90');
  const [amount, setAmount] = React.useState<number | null>(12.9);
  return (
    <View style={{ gap: 12 }}>
      <UI.AmountField
        label={de ? 'Betrag' : 'Amount'}
        accessibilityLabel={de ? 'Betrag' : 'Amount'}
        value={value}
        onChangeText={setValue}
        onAmountChange={setAmount}
        locale={de ? 'de-DE' : 'en-US'}
        currency="EUR"
      />
      <UI.Text accessibilityLiveRegion="polite">
        {amount === null
          ? de
            ? 'Unvollständiger Betrag'
            : 'Incomplete amount'
          : String(amount)}
      </UI.Text>
    </View>
  );
}

export function Example() {
  return (
    <GestureHandlerRootView style={{ flex: 1 }}>
      <SafeAreaProvider>
        <UI.IsolineProvider theme={UI.themes.violet.light}>
          <AmountFieldExample de={false} />
        </UI.IsolineProvider>
      </SafeAreaProvider>
    </GestureHandlerRootView>
  );
}
```
