Share visible labels, required markers, hints and errors across controls.

## Usage and states

Wrap controls that do not already render their own label/error. `requiredLabel` is visible localized text, not a validation rule. Pass the same label and error/hint to the child control for its accessible name and hint; this wrapper does not clone children or generate native label associations. Errors take precedence over hints and use a polite live region. Validation and submit state remain in the app.

## Complete example

The demo and this example use the same implementation. Reuse the existing provider when integrating this into an app.

```tsx title="FormfieldExample.tsx"
import * as React from 'react';
import { View, ScrollView } from 'react-native';
import { SafeAreaProvider } from 'react-native-safe-area-context';
import { GestureHandlerRootView } from 'react-native-gesture-handler';
import * as UI from '@isoline-ui/react-native';

type StoryProps = { de?: boolean };

function FormFieldExample({ de = false }: StoryProps) {
  const [checked, setChecked] = React.useState(false);
  const label = de ? 'Einwilligung' : 'Consent';
  const hint = de ? 'Du kannst die Auswahl ändern.' : 'You can change this choice.';
  return (
    <UI.FormField
      label={label}
      requiredLabel={de ? '(erforderlich)' : '(required)'}
      hint={hint}
    >
      <UI.Checkbox
        checked={checked}
        onCheckedChange={setChecked}
        accessibilityLabel={`${label}, ${hint}`}
        label={de ? 'Ich stimme zu' : 'I agree'}
      />
    </UI.FormField>
  );
}

export function Example() {
  return (
    <GestureHandlerRootView style={{ flex: 1 }}>
      <SafeAreaProvider>
        <UI.IsolineProvider theme={UI.themes.violet.light}>
          <FormFieldExample de={false} />
        </UI.IsolineProvider>
      </SafeAreaProvider>
    </GestureHandlerRootView>
  );
}
```
