Kontrollierte Auswahl mit aktivem, gemischtem und deaktiviertem Zustand.

## Vollständiges Beispiel

Die Demo und dieses Beispiel verwenden dieselbe Implementierung. In einer bestehenden App verwende den vorhandenen Provider weiter.

```tsx title="CheckboxExample.tsx"
import * as React from 'react';
import { View, ScrollView } from 'react-native';
import { SafeAreaProvider } from 'react-native-safe-area-context';
import { GestureHandlerRootView } from 'react-native-gesture-handler';
import * as UI from '@isoline-ui/react-native';

type StoryProps = { de?: boolean };

function CheckboxExample({ de = false }: StoryProps) {
  const [checked, setChecked] = React.useState(false);
  const [mixed, setMixed] = React.useState<boolean | 'mixed'>('mixed');
  const t = (en: string, german: string) => (de ? german : en);
  return (
    <View style={{ gap: 12 }}>
      <UI.Checkbox
        checked={checked}
        onCheckedChange={setChecked}
        accessibilityLabel={t('Select item', 'Eintrag auswählen')}
        label={t('Select item', 'Eintrag auswählen')}
      />
      <UI.Checkbox
        checked={mixed}
        onCheckedChange={setMixed}
        accessibilityLabel={t('Mixed selection', 'Teilweise ausgewählt')}
        label={t('Mixed selection', 'Teilweise ausgewählt')}
      />
      <UI.Checkbox
        checked
        disabled
        onCheckedChange={() => {}}
        accessibilityLabel={t('Unavailable selection', 'Nicht verfügbare Auswahl')}
        label={t('Unavailable selection', 'Nicht verfügbare Auswahl')}
      />
      <UI.Text>
        {checked ? t('Selected', 'Ausgewählt') : t('Not selected', 'Nicht ausgewählt')}
      </UI.Text>
    </View>
  );
}

export function Example() {
  return (
    <GestureHandlerRootView style={{ flex: 1 }}>
      <SafeAreaProvider>
        <UI.IsolineProvider theme={UI.themes.violet.light}>
          <CheckboxExample de={true} />
        </UI.IsolineProvider>
      </SafeAreaProvider>
    </GestureHandlerRootView>
  );
}
```

## API

Öffentliche Komponenten: `Checkbox`.

Wichtige Props: `checked`, `onCheckedChange`, `accessibilityLabel`, `label`, `disabled`. Die exportierten Props-Typen enthalten den vollständigen Vertrag. Beschriftungen und Aktionen werden von der App geliefert.

Prüfe große Schrift, deaktivierte Zustände und VoiceOver/TalkBack im nativen Bildschirm. Browser-Demos ersetzen keine Gerätetests.
