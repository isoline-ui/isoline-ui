Show static placeholders while content is loading.

## Usage and states

Use several blocks matching the final layout. Skeletons are static, non-interactive and hidden from assistive technology. Announce loading once on the surrounding region using a localized status and busy state. Replace placeholders when the request resolves; show an error and retry action on failure. Width/height accept React Native dimensions; `radius` is in dp.

## Complete example

The demo and this example use the same implementation. Reuse the existing provider when integrating this into an app.

```tsx title="SkeletonExample.tsx"
import * as React from 'react';
import { View, ScrollView } from 'react-native';
import { SafeAreaProvider } from 'react-native-safe-area-context';
import { GestureHandlerRootView } from 'react-native-gesture-handler';
import * as UI from '@isoline-ui/react-native';

type StoryProps = { de?: boolean };

function SkeletonExample({ de = false }: StoryProps) {
  const [loading, setLoading] = React.useState(true);
  return (
    <View style={{ gap: 16 }}>
      <UI.Button
        label={de ? 'Ladezustand wechseln' : 'Toggle loading'}
        onPress={() => setLoading(!loading)}
      />
      <View accessibilityState={{ busy: loading }}>
        {loading ? (
          <View style={{ gap: 12 }}>
            <UI.Text accessibilityLiveRegion="polite">
              {de ? 'Inhalt wird geladen' : 'Loading content'}
            </UI.Text>
            <UI.Skeleton width={64} height={64} radius={32} />
            <UI.Skeleton />
            <UI.Skeleton width="60%" />
          </View>
        ) : (
          <UI.ListItem title={de ? 'Inhalt geladen' : 'Content loaded'} />
        )}
      </View>
    </View>
  );
}

export function Example() {
  return (
    <GestureHandlerRootView style={{ flex: 1 }}>
      <SafeAreaProvider>
        <UI.IsolineProvider theme={UI.themes.violet.light}>
          <SkeletonExample de={false} />
        </UI.IsolineProvider>
      </SafeAreaProvider>
    </GestureHandlerRootView>
  );
}
```
